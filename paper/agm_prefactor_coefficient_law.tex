\begin{abstract}
For the lower Gauss--Legendre approximants $\pi_n$, consider the exact, reference-dependent diagnostic weight
$\alpha_n=-(\pi_{n+1}-\pi)/(\pi_{n+1}-\pi_n)$. Using the classical two-term theta-function error expansion recorded by Brent, we recover $\alpha_{n+1}/\alpha_n^2\to1/2$ as a direct corollary and sharpen the limit by an explicit rational prefactor model. With $x_n=2^{n+1}\pi$, we obtain
\[
\frac{\alpha_{n+1}}{\alpha_n^2}
=\frac{(4x_n-1)(x_n-1)^2}{(2x_n-1)^3}
+O(\operatorname{poly}(2^n)e^{-x_n})
=\frac12-\frac{3}{8x_n}+O(x_n^{-2}).
\]
This explains the observed approach to $1/2$ from below. The weight depends on the unknown target $\pi$ and is not an implementable acceleration method. We retain a separate numerical observation concerning numerator-channel weights, without claiming its proof.
\end{abstract}
\section{AGM indexing and the diagnostic coefficient}
Start with $a_0=1$, $b_0=1/\sqrt2$, $t_0=1/4$, and $p_0=1$, and use
\[
\begin{aligned}
a_{n+1}&=(a_n+b_n)/2,&b_{n+1}&=\sqrt{a_nb_n},\\
t_{n+1}&=t_n-p_n(a_n-a_{n+1})^2,&p_{n+1}&=2p_n.
\end{aligned}
\]
Throughout this paper the lower approximant is
\[
\pi_n=\frac{(a_n+b_n)^2}{4t_n}=\frac{a_{n+1}^2}{t_n},\qquad
\varepsilon_n=\pi_n-\pi<0.
\]
The index on $\pi_n$ is the index of $a_n,b_n,t_n$ appearing on the left expression, not of $a_{n+1}$. Define
\[
\alpha_n=-\frac{\pi_{n+1}-\pi}{\pi_{n+1}-\pi_n}
=-\frac{\varepsilon_{n+1}}{\varepsilon_{n+1}-\varepsilon_n}.
\]
Thus $\pi=\pi_{n+1}+\alpha_n(\pi_{n+1}-\pi_n)$ holds \emph{by definition}. Since evaluation of $\alpha_n$ requires $\pi$, this coefficient is a retrospective diagnostic, not a computable error-correction rule or algorithmic speed claim.
\section{Classical prefactor input}
We use the following detailed expansion from Brent's theta-function analysis of the lower AGM approximant \cite{brent2020}:
\begin{equation}\label{eq:brent}
\pi-\pi_n=(2^{n+4}\pi^2-8\pi)e^{-\pi 2^{n+1}}+O(2^n e^{-\pi 2^{n+2}}).
\end{equation}
The $O$ notation in \eqref{eq:brent} denotes a signed remainder with bounded magnitude; no sign is asserted for that remainder. This detailed two-term expansion, rather than the original 1976 complexity analysis \cite{brent1976}, is the asymptotic input used below.
Put $x_n=2^{n+1}\pi$ and $B_n=-8\pi(x_n-1)$, so that
\begin{equation}\label{eq:error}
\varepsilon_n=B_n e^{-x_n}+O(2^n e^{-2x_n}),\qquad x_{n+1}=2x_n.
\end{equation}
Writing $\varepsilon_n=A_n e^{-x_n}$ gives
\[
A_n=B_n+O(2^n e^{-x_n}),\qquad A_{n+1}/A_n\longrightarrow2.
\]
The limit comes from Brent's expansion; it is not an independent new AGM theorem.
\section{Coefficient law and sharpened correction}
\begin{lemma}\label{lem:ratios}
If $r_n=\varepsilon_{n+1}/\varepsilon_n\to0$, then
\[
\alpha_n=\frac{r_n}{1-r_n},\qquad
\frac{\alpha_{n+1}}{\alpha_n^2}=
\frac{\varepsilon_{n+2}\varepsilon_n^2}{\varepsilon_{n+1}^3}
\frac{(1-r_n)^2}{1-r_{n+1}}.
\]
\end{lemma}
\begin{proof}Both identities follow by substitution and cancellation.\end{proof}
\begin{theorem}[Prefactor coefficient limit]\label{thm:limit}
For the above lower Gauss--Legendre sequence,
\[\lim_{n\to\infty}\frac{\alpha_{n+1}}{\alpha_n^2}=\frac12.\]
\end{theorem}
\begin{proof}
By \eqref{eq:error}, $r_n\to0$. Lemma~\ref{lem:ratios} reduces the ratio asymptotically to $\varepsilon_{n+2}\varepsilon_n^2/\varepsilon_{n+1}^3$. The exponential factors cancel exactly because $x_{n+2}+2x_n-3x_{n+1}=0$. The remaining ratio tends to $2(1/2)^2=1/2$, since $A_{n+1}/A_n\to2$.
\end{proof}
\begin{corollary}[Finite-index rational prefactor law]\label{cor:sharp}
Let $x=x_n=2^{n+1}\pi$. Then
\begin{equation}\label{eq:rational}
\frac{\alpha_{n+1}}{\alpha_n^2}
=\frac{(4x-1)(x-1)^2}{(2x-1)^3}
+O(\operatorname{poly}(2^n)e^{-x}),
\end{equation}
and hence
\begin{equation}\label{eq:expansion}
\frac{\alpha_{n+1}}{\alpha_n^2}
=\frac12-\frac{3}{8x}+O(x^{-2})
=\frac12-\frac{3}{2^{n+4}\pi}+O(4^{-n}).
\end{equation}
Also the rescaled error prefactors obey
\[
\frac{A_{n+1}}{A_n}=2+\frac1{x-1}+O(\operatorname{poly}(2^n)e^{-x}).
\]
\end{corollary}
\begin{proof}
Use $B_n=-8\pi(x-1)$, $B_{n+1}=-8\pi(2x-1)$, and $B_{n+2}=-8\pi(4x-1)$ in Lemma~\ref{lem:ratios}. The pure prefactor quotient is exactly the rational expression in \eqref{eq:rational}. By \eqref{eq:error}, the relative errors in the $B_k$ are exponentially small, and $r_n=O(e^{-x}\operatorname{poly}(2^n))$; their effect on the diagnostic quotient is likewise exponentially small. Expansion in $1/x$ gives $1/2-3/(8x)+O(x^{-2})$. The prefactor ratio follows from $B_{n+1}/B_n=(2x-1)/(x-1)$.
\end{proof}
\begin{remark}
Equation \eqref{eq:expansion} describes the eventual approach to $1/2$ from below. The finite-$n$ rational expression is an asymptotic model, not an exact identity for the numerical AGM coefficient. Exponentially small corrections remain.
\end{remark}

\section{Numerator channel and computational scope}
Set $q_n=(a_n+b_n)^2$ and
\[\beta_n=-\frac{q_{n+1}-4\pi t_{n+1}}{q_{n+1}-q_n}.\]
\begin{observation}[Numerical only]
The original high-precision calculation suggested $\beta_{n+1}/\beta_n^2\to0$. No analytical proof is provided, and the present revision does not claim independent reproduction of this secondary observation.
\end{observation}
The static Baker N-series uses fixed rational weights to remove powers of a resolution parameter. In contrast the present $\alpha_n$ uses an unknown limit and is \emph{not} an acceleration prescription. Aitken's $\Delta^2$ extrapolation is a different, implementable three-point operation; its merit relative to taking the next AGM iterate is not asserted here without a separate error-and-cost comparison.
\section{Conclusion}
The limit $\alpha_{n+1}/\alpha_n^2\to1/2$ is a transparent consequence of Brent's lower-approximant asymptotic. Retaining his subleading $-8\pi$ prefactor term yields the sharper rational model \eqref{eq:rational}, which quantitatively explains the finite-index trend. This is a diagnostic comparison of error structure, not a new algorithm for $\pi$ or a fixed-coefficient extrapolation method.
